function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('🛡️ 안전 패트롤 관리')
    .addItem('시트 및 헤더 자동 초기화/생성', 'setupSheetsAuto')
    .addItem('시트1 샘플 작업 데이터 등록', 'injectSamplePermits')
    .addToUi();
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet1 = ss.getSheetByName("시트1");
  if (!sheet1) { setupSheetsAuto(); sheet1 = ss.getSheetByName("시트1"); }
  var data = sheet1.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({ status: "success", count: Math.max(0, data.length - 1), data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet2 = ss.getSheetByName("시트2");
    if (!sheet2) { setupSheetsAuto(); sheet2 = ss.getSheetByName("시트2"); }
    var payload = JSON.parse(e.postData.contents);
    if (payload.action === "saveInspection" && payload.row) {
      sheet2.appendRow(payload.row);
      var lastRow = sheet2.getLastRow();
      sheet2.setRowHeight(lastRow, 28);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", rowNumber: lastRow }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
